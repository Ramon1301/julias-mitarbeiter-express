// Spiel-Kachel: Cover + wer gerade im Highscore vorne liegt
let gameBoard = {};
async function loadGameBoard() {
  try { const t = await (await fetch(STORE_URL + '-spiel?t=' + Date.now(), { cache: 'no-store' })).text(); gameBoard = t.trim() ? JSON.parse(t) : {}; if (view === 'dash') render(); } catch {}
}
function gameTile() {
  const lead = Object.entries(gameBoard).sort((a, b) => b[1].best - a[1].best)[0];
  return `<a class="tile game" href="schlechte-zeiten.html?v=${Date.now()}">
      <img class="cover" src="assets/johann-wut.jpg" alt="Johann weint auf der Bühne, das Publikum buht">
      <div class="game-info">
        <div><h3>Für die schlechten Zeiten</h3><p>Bewirf Johann mit Tomaten, Eiern, Torten & Melonen</p></div>
        <span class="go">Spielen →</span>
      </div>
      ${lead ? `<div class="next" data-p="${esc(lead[0])}">${avatar(lead[0])}<span>🏆 <b>${esc(lead[0])}</b> führt mit ${lead[1].best} Punkten</span></div>` : '<div class="next"><span>🏆 Noch kein Highscore – wer fängt an?</span></div>'}
    </a>`;
}
